defined_text = {
    name = GetSndConstructionStatus

        trigger = { has_country_flag = snd_external_construction }
        localization_key = snd_construction_status_external
    }
    text = {
        trigger = { NOT = { has_country_flag = snd_external_construction } }
        localization_key = snd_construction_status_available
    }
}

defined_text = {
    name = GetSndAutoConstructionStatus
    text = {
        trigger = { NOT = { has_country_flag = snd_auto_construction } }
        localization_key = snd_auto_construction_off
    }
    text = {
        trigger = {
            has_country_flag = snd_auto_construction
            has_country_flag = snd_external_construction
        }
        localization_key = snd_auto_construction_suspended
    }
    text = {
        trigger = {
            has_country_flag = snd_auto_construction
            NOT = { has_country_flag = snd_external_construction }
            has_country_flag = snd_auto_construction_cooldown
        }
        localization_key = snd_auto_construction_waiting
    }
    text = {
        trigger = {
            has_country_flag = snd_auto_construction
            NOT = { has_country_flag = snd_external_construction }
            NOT = { has_country_flag = snd_auto_construction_cooldown }
        }
        localization_key = snd_auto_construction_active
    }
}
