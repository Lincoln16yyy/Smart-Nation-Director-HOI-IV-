snd_dispatch_pilot = {
    capital_scope = {
        add_building_construction = {
            type = infrastructure
            level = 1
            instant_build = no
        }
    }
    set_country_flag = snd_pilot_dispatched
    clr_country_flag = snd_pilot_enabled
    country_event = { id = snd.1 }
}

snd_capture_industry_baseline = {
    set_variable = { snd_previous_civilian_factories = num_of_civilian_factories }
    set_variable = { snd_previous_military_factories = num_of_military_factories }
}

snd_update_industry_tracking = {
    set_variable = { snd_civilian_factory_change = num_of_civilian_factories }
    subtract_from_variable = { snd_civilian_factory_change = snd_previous_civilian_factories }
    set_variable = { snd_military_factory_change = num_of_military_factories }
    subtract_from_variable = { snd_military_factory_change = snd_previous_military_factories }
    snd_capture_industry_baseline = yes
    set_country_flag = snd_industry_sample_ready
}

snd_dispatch_auto_construction = {
    if = {
        limit = { capital_scope = { snd_auto_factory_candidate = yes } }
        capital_scope = { snd_queue_preferred_factory = yes }
    }
    else_if = {
        limit = { any_owned_state = { snd_auto_factory_candidate = yes } }
        random_owned_controlled_state = {
            limit = { snd_auto_factory_candidate = yes }
            snd_queue_preferred_factory = yes
        }
    }
    else_if = {
        limit = {
            snd_pilot_capital_eligible = yes
            capital_scope = { is_demilitarized_zone = no }
        }
        capital_scope = {
            ROOT = {
                set_variable = { snd_auto_project_type = 3 }
                set_variable = { snd_auto_project_baseline = PREV.building_level@infrastructure }
            }
            add_building_construction = {
                type = infrastructure
                level = 1
                instant_build = no
            }
            snd_record_auto_project = yes
        }
    }
}

snd_queue_preferred_factory = {
    if = {
        limit = { ROOT = { has_war = yes } }
        ROOT = {
            set_variable = { snd_auto_project_type = 2 }
            set_variable = { snd_auto_project_baseline = PREV.building_level@arms_factory }
        }
        add_building_construction = {
            type = arms_factory
            level = 1
            instant_build = no
        }
    }
    else = {
        ROOT = {
            set_variable = { snd_auto_project_type = 1 }
            set_variable = { snd_auto_project_baseline = PREV.building_level@industrial_complex }
        }
        add_building_construction = {
            type = industrial_complex
            level = 1
            instant_build = no
        }
    }
    snd_record_auto_project = yes
}

snd_record_auto_project = {
    ROOT = {
        set_variable = { snd_auto_project_state = PREV }
        set_country_flag = snd_auto_project_pending
        set_country_flag = { flag = snd_auto_construction_cooldown days = 180 }
    }
}

snd_capture_report_values = {
    set_variable = { snd_report_civilian_factories = num_of_civilian_factories }
    set_variable = { snd_report_military_factories = num_of_military_factories }
    set_variable = { snd_report_dockyards = num_of_naval_factories }
    set_variable = { snd_report_manpower_k = manpower_k }
}

snd_show_national_report = {
    snd_capture_report_values = yes
    country_event = { id = snd.3 }
}

snd_refresh_dashboard = {
    snd_capture_report_values = yes
}

snd_review_national_risk = {
    set_temp_variable = { snd_current_risk = 0 }
    if = {
        limit = { snd_has_stability_risk = yes }
        set_temp_variable = { snd_current_risk = 1 }
    }
    else_if = {
        limit = { snd_has_war_support_risk = yes }
        set_temp_variable = { snd_current_risk = 2 }
    }
    else_if = {
        limit = { snd_has_manpower_risk = yes }
        set_temp_variable = { snd_current_risk = 3 }
    }
    else_if = {
        limit = { snd_has_infantry_equipment_risk = yes }
        set_temp_variable = { snd_current_risk = 4 }
    }
    if = {
        limit = {
            NOT = { check_variable = { snd_last_reported_risk = snd_current_risk } }
        }
        set_variable = { snd_last_reported_risk = snd_current_risk }
        if = {
            limit = { check_variable = { snd_current_risk > 0 } }
            snd_show_national_report = yes
        }
    }
}

snd_start_industry_tracking = {
    snd_capture_industry_baseline = yes
    clr_country_flag = snd_industry_sample_ready
    set_country_flag = snd_industry_tracking
}

snd_stop_industry_tracking = {
    clr_country_flag = snd_industry_tracking
    clr_country_flag = snd_industry_sample_ready
}

snd_start_risk_alerts = {
    set_variable = { snd_last_reported_risk = 0 }
    set_country_flag = snd_risk_alerts_enabled
}

snd_start_government_routines = {
    if = {
        limit = { NOT = { has_country_flag = snd_industry_tracking } }
        snd_start_industry_tracking = yes
    }
    if = {
        limit = { NOT = { has_country_flag = snd_risk_alerts_enabled } }
        snd_start_risk_alerts = yes
    }
    if = {
        limit = { snd_can_start_auto_construction = yes }
        set_country_flag = snd_auto_construction
    }
}
