on_actions = {
    on_monthly = {
        effect = {
            if = {
                limit = {
                    is_ai = no
                    has_country_flag = snd_industry_tracking
                }
                snd_update_industry_tracking = yes
            }
            if = {
                limit = {
                    is_ai = no
                    has_country_flag = snd_pilot_enabled
                    NOT = { snd_construction_managed_externally = yes }
                    NOT = { has_country_flag = snd_pilot_dispatched }
                    snd_pilot_capital_eligible = yes
                }
                snd_dispatch_pilot = yes
            }
            if = {
                limit = {
                    is_ai = no
                    has_country_flag = snd_auto_project_pending
                    snd_auto_project_level_increased = yes
                }
                clr_country_flag = snd_auto_project_pending
            }
            if = {
                limit = {
                    is_ai = no
                    has_country_flag = snd_auto_construction
                    NOT = { snd_construction_managed_externally = yes }
                    NOT = { has_country_flag = snd_auto_construction_cooldown }
                    NOT = { has_country_flag = snd_auto_project_pending }
                    NOT = { has_country_flag = snd_pilot_enabled }
                }
                snd_dispatch_auto_construction = yes
            }
            if = {
                limit = {
                    is_ai = no
                    has_country_flag = snd_risk_alerts_enabled
                }
                snd_review_national_risk = yes
            }
        }
    }
}
